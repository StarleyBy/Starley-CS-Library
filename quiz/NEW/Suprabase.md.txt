1. supabase login / supabase link — деплой Edge Function
bash
# Установить CLI (один из вариантов)
npm install -g supabase
# либо: brew install supabase/tap/supabase   (mac)
# либо: scoop install supabase               (windows)

# Логин — откроет браузер для OAuth-авторизации
supabase login

# В корне репозитория: если папки supabase/ ещё нет — инициализировать
supabase init

# Привязать локальную папку к проекту (project-ref смотрите в
# Settings → General → Reference ID в дашборде Supabase)
supabase link --project-ref <project-ref>

# Задеплоить функцию (функция уже лежит в supabase/functions/admin-create-user/)
supabase functions deploy admin-create-user
